import { Card, CardGrid } from '@astrojs/starlight/components'

<CardGrid>
  <Card title="Un appel, sans préalable" icon="rocket">
    Ni clé, ni jeton, ni compte. La première requête part du terminal.

    ```sh wrap
    curl 'https://api-equides.org/v1/search?q=qab'
    ```
  </Card>

  <Card title="Le graphe en une requête" icon="approve-check">
    Un pedigree sur cinq générations touche 31 ancêtres : un seul appel,
    au lieu de 31 allers-retours.

    ```sh wrap
    curl 'https://api-equides.org/v1/equides/{id}/pedigree?generations=5'
    ```
  </Card>

  <Card title="Des facettes sans les lignes" icon="bars">
    Ventiler une sélection ne rapatrie que des effectifs, jamais les fiches.

    ```sh wrap
    curl 'https://api-equides.org/v1/stats/repartition?dimension=race'
    ```
  </Card>

  <Card title="Service indépendant" icon="warning">
    Non affilié à l'IFCE. La donnée servie est une extraction datée ; les
    fiches en ligne font seules autorité.

    [Provenance et licence](/donnees/provenance/)
  </Card>
</CardGrid>
